### 迪杰斯特拉

无负权环的单源最短路算法，时间复杂度为
$$
O(m\log_2n)
$$

```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long
typedef pair<int,int> PII;
const int N=1e5+7;
int dis[N];
int pre[N];
bool vis[N];
vector<PII> e[N];

void dij(int s){
	priority_queue<PII,vector<PII>,greater<PII>> pq;
	pq.push({0,s});
	dis[s]=0;
	pre[s]=s;
	while(!pq.empty()){
		int u=pq.top().second;
		pq.pop();
		if(vis[u]) continue;
		vis[u]=1;
		for(auto &p:e[u]){
			int v=p.first;
			int w=p.second;
			if(dis[u]+w<dis[v]){
				pre[v]=u;
				dis[v]=dis[u]+w;
				pq.push({dis[v],v});
			}
		}
	}
}
void print(int x){
	if(pre[x]==x){
		return;
	}
	x=pre[x];
	print(x);
	cout<<x<<"->";
}
void solve(){
    int n,m,s;
	cin>>n>>m>>s;
	while(m--){
		int u,v,w;
		cin>>u>>v>>w;
		e[u].push_back({v,w});
	}
	for(int i=1;i<=n;i++){
		vis[i]=0;
		dis[i]=INT_MAX;
	}
	dij(s);
	for(int i=1;i<=n;i++){
		cout<<dis[i]<<" ";
		//print(i);
		//cout<<i<<'\n';
	}
}
signed main(){
	ios::sync_with_stdio(0);
    cin.tie(0);
	int T=1;
	//cin>>T;
	while(T--){
		solve();
	}
	return 0;
}
```

### floyd

多源最短路算法，时间复杂度为$O(n^3)$

```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long

int dist[107][107];
int n,m;

void floyd(){
    // k：枚举中转点
    for(int k=1;k<=n;k++){
        // i：枚举起点
        for(int i=1;i<=n;i++){
            // j：枚举终点
            for(int j=1;j<=n;j++){
                // 松弛操作：i->j 直接走 或 i->k->j 中转，取更短的
                dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);
            }
        }
    }
}

signed main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            dist[i][j]=INT_MAX;
            if(i==j) dist[i][j]=0;
        }
    }
    while(m--){
        int u,v,w;
        cin>>u>>v>>w;
        dist[u][v]=min(dist[u][v],w);
        dist[v][u]=dist[u][v];
    }
    floyd();
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            cout<<dist[i][j]<<' ';
        }
        cout<<'\n';
    }
    return 0;
}
```

### Bellman-Ford

存在负权边的单源最短路问题，可以判断从起点 `s` 出发能否遇到负环，时间复杂度为$O(mn)$

```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long
const int N=1e4+7;
int dis[N];
int n,m;

struct edge{
    int u,v,w;
};
vector<edge> e;

// 功能：1.求起点s到所有点的最短路径 2.检测图中是否存在负权环
// 返回值：true=有负权环，false=无负权环
bool BellmanFord(int s){
    // 一开始所有点都不可达，距离设为无穷大
    for(int i=1;i<=n;i++){
        dis[i]=INT_MAX;
    }
    // 起点到自己的距离为 0
    dis[s]=0;
    // 进行 n-1 轮松弛操作
    // 原理：不含负环的图中，两点间的最短路径最多经过 n-1 条边
    for(int i=1;i<=n-1;i++){
        bool update=false; // 标记本轮循环是否更新了距离
        // 遍历图中的每一条边，尝试松弛
        for(auto &p:e){
            // 必须保证起点 u 是可达的（dis[u]不是无穷大），才能松弛
            if(dis[p.u]!=INT_MAX){
                // 松弛操作：如果 s->u->v 比 s->v 更短，就更新
                if(dis[p.u] + p.w < dis[p.v]){
                    dis[p.v] = dis[p.u] + p.w;
                    update = true; // 标记本轮有更新
                }
            }
        }
        // 优化：如果本轮没有任何更新，说明最短路径已经全部找到，提前退出
        if(!update){
            break;
        }
    }

    // 如果第 n 次还能松弛，说明路径可以无限缩短 → 存在负权环
    for(auto &p:e){
        if(dis[p.u]!=INT_MAX){
            // 还能更新距离 → 发现负权环
            if(dis[p.u] + p.w < dis[p.v]){
                return true;
            }
        }
    }

    // 遍历完所有边都没有更新 → 没有负权环
    return false;
}

signed main(){
    int start=1; // 默认起点为1号节点
    cin>>n>>m;
    // 读入m条有向边
    while(m--){
        int u,v,w;
        cin>>u>>v>>w;
        e.push_back({u,v,w});   
    }

    // 调用算法，判断是否有负权环
    if(BellmanFord(start)){
        cout<<"YES"<<'\n';// 输出YES：图中存在负权环
    }else{
        cout<<"NO"<<'\n';// 输出NO：图中无负权环
        // 输出起点到每个节点的最短距离
        for(int i=1;i<=n;i++){
            cout<<dis[i]<<' ';
        }
    }
    return 0;
}
```

### SPFA

存在负权边但无负权环的单源最短路问题，时间复杂度最好为$O(m)$，最坏为$O(mn)$

```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long
typedef pair<int,int> PII;
const int N=1e5+7;

int dis[N];   // 存储起点到各点最短距离
int cnt[N];   // 记录起点到当前点经过的边数
bool vis[N];  // 标记节点是否在队列中
int n,m,s;    // n点数 m边数 s起点
vector<PII> e[N]; // 邻接表存图

// SPFA 判断是否存在负环
bool SPFA(int s){
    queue<int> q;
    dis[s]=0;        // 起点距离初始为0
    q.push(s);       // 起点入队
    vis[s]=1;        // 标记起点在队列中
    while(!q.empty()){
        int u=q.front();
        q.pop();
        vis[u]=0;    // 出队，取消标记
        // 遍历u的所有邻边
        for(auto &p:e[u]){
            int v=p.first;
            int w=p.second;
            // 防止INT_MAX加法溢出，松弛更新最短路
            if(dis[u]!=INT_MAX&&dis[u]+w<dis[v]){
                dis[v]=dis[u]+w;
                // 更新到v经过的边数
                cnt[v]=cnt[u]+1;
                // 经过边数>=点数，必然存在负环
                if(cnt[v]>=n) return true;
                // 不在队列则入队
                if(!vis[v]){
                    vis[v]=1;
                    q.push(v);
                }
            }
        }
    }
    return false; // 无负环
}

void solve(){
    cin>>n>>m>>s;
    // 多组样例清空邻接表
    for(int i=1;i<=n;i++){
        e[i].clear();
    }
    // 初始化距离、标记、边数数组
    for(int i=1;i<=n;i++){
        dis[i]=INT_MAX;
        vis[i]=0;
        cnt[i]=0;
    }
    // 建边
    while(m--){
        int u,v,w;
        cin>>u>>v>>w;
        if(w>=0){
            e[u].push_back({v,w});
            e[v].push_back({u,w});
        }else{
            e[u].push_back({v,w});
        }
    }
    // 负环判定输出
    if(SPFA(s)){
        cout<<"YES"<<'\n'; // 存在负环
    }else{
        cout<<"NO"<<'\n';  // 无负环
        //for(int i=1;i<=n;i++){
        //    cout<<dis[i]<<' ';
        //}
    }
}

signed main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    int T=1;
    cin>>T;
    while(T--){
        solve();
    }
}
```

### 拓扑排序
```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long
typedef pair<int,int> PII;
const int mod=1e9+7;

int degree[1000];
vector<int> e[1000];
vector<int> ans;
int n,m;

bool bfs(){
    queue<int> q;
    int cnt=0;
    for(int i=1;i<=n;i++){
        if(degree[i]==0){
            q.push(i);
        }
    }
    while(!q.empty()){
        int u=q.front();
        q.pop();
        ans.push_back(u);
        cnt++;
        for(int v:e[u]){
            degree[v]--;
            if(degree[v]==0){
                q.push(v);;
            }
        }
    }
    if(cnt==n) return true;
    else return false;
}

void solve(){
    cin>>n>>m;
    while(m--){
        int u,v;
        cin>>u>>v;
        e[u].push_back(v);
        degree[v]++;
    }
    if(bfs()){
        cout<<"Yes"<<'\n';//无环，可以拓扑排序
        for(int p:ans){
            cout<<p<<' ';
        }
    }else{
        cout<<"No"<<'\n';//有环不能拓扑排序
    }
}

signed main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    int T=1;
    //cin>>T;
    while(T--){
        solve();
    }
    return 0;
}
```

### Kruskal
```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long
#define all(x) x.begin(),x.end()
#define YES cout<<"YES\n"
#define NO cout<<"NO\n"
#define cfs cout<<fixed<<setprecision
typedef pair<int,int>PII;
const int N=2e5+10;
const int INF=1e18;
int n,m,s;
int p[N];

// 边结构体：存 u->v，权值 w
struct node{
    int u;
    int v;
    int w;
    // 重载小于号：按边权从小到大排序
    bool operator<(const node &other)const{
        return w<other.w;
    }
};
// 并查集找根（带路径压缩）
int find(int x){
    if(p[x]!=x) p[x]=find(p[x]);
    return p[x];
}
// 合并两个集合
void merge(int x,int y){
    int rx=find(x),ry=find(y);
    if(rx!=ry) p[rx]=ry;
}
void solve(){
    int n,m;
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        p[i]=i;
    }
    vector<node> edge;
    while(m--){
        int u,v,w;
        cin>>u>>v>>w;
        edge.push_back({u,v,w});
    }
    sort(all(edge));

    int sum=0;  // 最小生成树总权值
    int cnt=0;  // 已选边数
    vector<node> ans; // 存选中的边

    // 2. 从小到大枚举每条边
    for(auto p:edge){
        int u=p.u;
        int v=p.v;
        int w=p.w;
        // 如果 u 和 v 已经连通 → 加这条边会成环 → 跳过
        if(find(u)==find(v)) continue;
        // 不连通 → 选这条边
        merge(u,v);       // 合并两个集合
        sum += w;         // 累加权值
        ans.push_back(p); // 记录这条边
        cnt++;            // 已选边数+1
        // 最小生成树一共需要 n-1 条边，够了就退出
        if(cnt==n-1) break;
    }
    // 选够 n-1 条边 → 生成树存在，输出总权值
    if(cnt==n-1){
        cout<<sum<<'\n';
        for(auto p:ans){
        cout<<p.u<<' '<<p.v<<' '<<p.w<<'\n';
    	}
    }
    // 不够 → 图不连通，无法构成生成树
    else cout<<"orz"<<'\n';
}

signed main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    int T=1;
    while(T--) solve();
    return 0;
}
```

```cpp
const int N=1e5+5;

int haxi[N]; // 前缀哈希数组
int p[N];    // 进制的幂次数组

// 预处理字符串哈希与幂次数组
int get_haxi(string s,int b,int mod){
    int lens=s.size();
    s=" "+s; // 转为1下标
    for(int i=1;i<=lens;i++){
        haxi[i]=(haxi[i-1]*b%mod+s[i]%mod)%mod;
        p[i]=p[i-1]*b%mod;
    }
    return haxi[lens];
}

// 获取子串 s[l..r] 的哈希值
int get_lr(int l,int r,int mod){
    return (haxi[r]-haxi[l-1]*p[r-l+1]%mod+mod)%mod;
```

```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long

signed main(){
    int n,m;
    // n物品个数 m背包容量
    cin>>n>>m;
    // w重量 v价值 dp[j]容量j的最大价值
    vector<int> w(n),v(n),dp(m+1,0);

    for(int i=0;i<n;i++){
        cin>>w[i]>>v[i];

    return 0;
```

### 组合数
```cpp

#define all(x) x.begin(),x.end()
#define YES cout<<"YES\n"
#define NO cout<<"NO\n"

const int INF=1e18;
const int mod=1e9+7;

int fact[N],invfact[N];//阶乘数组fact与逆元阶乘数组invfact

int qpow(int a,int x){
    int ans=1;
    while(x){
        if(x&1) ans=ans*a%mod;
        a=a*a%mod;
        x>>=1;
    }
    return ans;
}

void init(){
    //预处理阶乘数组fact与逆元阶乘数组invfact，用于组合数计算
    fact[0]=1;
    //递推计算0~N-1的阶乘，全程取模防止溢出
    for(int i=1;i<N;i++){
        fact[i]=fact[i-1]*i%mod;
    }
    //费马小定理求最大下标阶乘的逆元，mod为质数
    invfact[N-1]=qpow(fact[N-1],mod-2);
    //倒序递推所有阶乘逆元，利用inv[i]=inv[i+1]*(i+1)%mod公式
    for(int i=N-2;i>=0;i--){
        invfact[i]=invfact[i+1]*(i+1)%mod;
    }
}

    //组合数公式c(n,m)=n!/(m!*(n-m)!)，利用预处理逆元阶乘转化为乘法取模
    return fact[n]*invfact[m]%mod*invfact[n-m]%mod;
}

    init();
    int n,m;
    cin>>n>>m;

}
```

### 单调栈

遍历1-n找的是左边第一个比它大/小的数，遍历n-1则找的是右边第一个比它大/小的数

```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long

void solve(){
    int n;
    cin>>n;
    vector<int> a(n+1,0);    // a数组存储输入序列，下标从1开始
    vector<int> ans1(n+1,0); // ans1存储每个元素左侧第一个比它大的元素下标
    vector<int> ans2(n+1,0); // ans2存储每个元素左侧第一个比它小的元素下标
    stack<int> st;           // 单调栈，栈内存放数组下标，维护单调性
    
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    // 单调递减栈：求左侧第一个严格大于当前a[i]的位置，若要找大于等于的则将while里<=改为<即可
    for(int i=1;i<=n;i++){
        // 栈不为空 且 栈顶对应数值 <= 当前值，弹出（不满足"更大"条件）
        while(!st.empty()&&a[st.top()]<=a[i]) st.pop();
        // 栈空说明左边没有更大值，存0；否则存栈顶下标
        ans1[i]=st.empty()?0:st.top();
        st.push(i); // 当前下标入栈，供后续元素比较
    }
    // 输出左侧第一个更大元素下标结果
    for(int i=1;i<=n;i++){
        cout<<ans1[i]<<' ';
    }
    cout<<'\n';
    // 清空栈，准备下一轮单调栈计算
    while(!st.empty()) st.pop();
    // 单调递增栈：求左侧第一个严格小于当前a[i]的位置，若要找小于等于的则将while里>=改为>即可
    for(int i=1;i<=n;i++){
        // 栈不为空 且 栈顶对应数值 >= 当前值，弹出（不满足"更小"条件）
        while(!st.empty()&&a[st.top()]>=a[i]) st.pop();
        // 栈空说明左边没有更小值，存0；否则存栈顶下标
        ans2[i]=st.empty()?0:st.top();
        st.push(i); // 当前下标入栈
    }
    // 输出左侧第一个更小元素下标结果
    for(int i=1;i<=n;i++){
        cout<<ans2[i]<<' ';
    }
    cout<<'\n';

}
```
